MODELO LÓGICO - CLINICA_CARE
=============================
Convenção: PK = Primary Key | FK = Foreign Key | UK = Unique Key

--------------------------------------------------------
TABELA: pacientes
--------------------------------------------------------
id_paciente         INT             PK, AUTO_INCREMENT
nome_completo       VARCHAR(150)    NOT NULL
cpf                 VARCHAR(14)     NOT NULL, UK
data_nascimento     DATE            NOT NULL
genero              VARCHAR(20)
endereco            VARCHAR(255)
telefone            VARCHAR(20)
email               VARCHAR(150)
tipo_plano          VARCHAR(50)     NOT NULL  -- particular, convenio_x, convenio_y
data_cadastro       DATE            NOT NULL, DEFAULT CURRENT_DATE

--------------------------------------------------------
TABELA: medicos
--------------------------------------------------------
id_medico               INT             PK, AUTO_INCREMENT
nome_completo            VARCHAR(150)    NOT NULL
crm                      VARCHAR(20)     NOT NULL, UK
telefone                 VARCHAR(20)
email                    VARCHAR(150)
data_contratacao         DATE
status_ativo             BOOLEAN         DEFAULT TRUE
horario_disponibilidade  VARCHAR(255)    -- ex: "Seg-Sex 08h-12h"

--------------------------------------------------------
TABELA: especialidades
--------------------------------------------------------
id_especialidade      INT             PK, AUTO_INCREMENT
nome_especialidade     VARCHAR(100)    NOT NULL, UK
descricao               VARCHAR(255)
valor_base_consulta     DECIMAL(10,2)   NOT NULL
duracao_media_minutos   INT
ativa                    BOOLEAN         DEFAULT TRUE
data_criacao             DATE            DEFAULT CURRENT_DATE
observacoes              VARCHAR(255)

--------------------------------------------------------
TABELA: medico_especialidade  (tabela associativa - resolve N:N)
--------------------------------------------------------
id_medico          INT     PK, FK -> medicos(id_medico)
id_especialidade   INT     PK, FK -> especialidades(id_especialidade)
data_inicio        DATE
principal          BOOLEAN DEFAULT FALSE
-- PK composta (id_medico, id_especialidade)

--------------------------------------------------------
TABELA: consultas
--------------------------------------------------------
id_consulta          INT             PK, AUTO_INCREMENT
id_paciente          INT             NOT NULL, FK -> pacientes(id_paciente)
id_medico            INT             NOT NULL, FK -> medicos(id_medico)
id_especialidade     INT             NOT NULL, FK -> especialidades(id_especialidade)
data_consulta        DATE            NOT NULL
hora_consulta        TIME            NOT NULL
status                VARCHAR(20)     NOT NULL, DEFAULT 'agendada'  -- agendada, realizada, cancelada, faltou
valor_consulta        DECIMAL(10,2)   NOT NULL
observacoes           VARCHAR(255)
data_agendamento      DATE            NOT NULL, DEFAULT CURRENT_DATE

--------------------------------------------------------
TABELA: prontuarios
--------------------------------------------------------
id_prontuario        INT     PK, AUTO_INCREMENT
id_consulta          INT     NOT NULL, UK, FK -> consultas(id_consulta)   -- UK garante 1:1
data_registro         DATE    NOT NULL
diagnostico            TEXT
anotacoes               TEXT
sintomas_relatados      TEXT
exames_solicitados      TEXT
observacoes             TEXT

--------------------------------------------------------
TABELA: prescricoes
--------------------------------------------------------
id_prescricao        INT             PK, AUTO_INCREMENT
id_prontuario         INT             NOT NULL, FK -> prontuarios(id_prontuario)
medicamento            VARCHAR(150)    NOT NULL
dosagem                VARCHAR(50)
frequencia             VARCHAR(50)
duracao_tratamento     VARCHAR(50)
data_prescricao        DATE            NOT NULL
observacoes            VARCHAR(255)

--------------------------------------------------------
TABELA: pagamentos
--------------------------------------------------------
id_pagamento         INT             PK, AUTO_INCREMENT
id_consulta           INT             NOT NULL, UK, FK -> consultas(id_consulta)  -- UK garante 1:1
valor                  DECIMAL(10,2)   NOT NULL
data_pagamento         DATE
metodo_pagamento       VARCHAR(20)     -- dinheiro, cartao, pix
status_pagamento       VARCHAR(20)     NOT NULL, DEFAULT 'pendente'  -- pendente, pago, cancelado
data_vencimento        DATE
numero_recibo          VARCHAR(50)

--------------------------------------------------------
REGRAS DE INTEGRIDADE REFERENCIAL (ON DELETE)
--------------------------------------------------------
pacientes/medicos -> consultas: RESTRICT (não permite apagar paciente/médico com histórico)
consultas -> prontuarios: RESTRICT
consultas -> pagamentos: RESTRICT
prontuarios -> prescricoes: CASCADE (remove prescrições junto se o prontuário for removido)
medicos/especialidades -> medico_especialidade: CASCADE
